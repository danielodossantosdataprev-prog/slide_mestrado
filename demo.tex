% 3
\begin{frame}{Motivação do problema}
  \begin{itemize}
    \item Na insuficiência cardíaca, a morte ocorre por duas vias:
      \begin{itemize}
        \item \textbf{falência de bomba}: deterioração progressiva;
        \item \textbf{morte súbita cardíaca (MSC)}: evento abrupto, em geral por
          arritmia ventricular, em paciente muitas vezes estável.
      \end{itemize}
    \item A MSC responde por cerca de \textbf{metade} das mortes nessa
      população \cite{tomaselli2004sudden}.
    \item Prevenção: o \textbf{cardiodesfibrilador implantável}, que reduziu a
      mortalidade \cite{bardy2005amiodarone}.
    \item A indicação baseia-se na \textbf{fração de ejeção (FEVE)}, critério
      imperfeito \cite{zeppenfeld2022esc}:
      \begin{itemize}
        \item muitos pacientes com o dispositivo nunca o utilizam;
        \item parte das mortes súbitas ocorre com FEVE acima do limiar.
      \end{itemize}
  \end{itemize}
  \msg{Há espaço para ferramentas que estratifiquem melhor o risco de MSC.}
\end{frame}

% 4
\begin{frame}{Modelos preditivos para a morte súbita cardíaca}
  \begin{columns}[T]
    \column{0.55\textwidth}
    \begin{itemize}
      \item Estimam, para cada paciente, a \textbf{probabilidade} do evento em
        um horizonte definido.
      \item Permitem ordenar os pacientes por risco, em vez de depender de um
        único limiar clínico.
      \item O \textbf{Holter de 24 horas} é não invasivo, barato e amplamente
        disponível.
      \item Capta ectopias, pausas e variações da frequência entre sono e
        vigília.
    \end{itemize}
    \column{0.43\textwidth}
    \centering
    \begin{tikzpicture}[node distance=0.35cm]
      \node[caixa, text width=3.6cm] (h) {Holter de 24 h\\3 derivações, 200 Hz};

\end{frame}

\begin{frame}{O que a literatura tem feito}
  \begin{columns}[T]
    \column{0.49\textwidth}
    \begin{block}{Marcadores definidos por especialistas}
      \small
      \begin{itemize}
        \item VFC, ectopia, turbulência, dinâmica da repolarização.
        \item Combinados com variáveis clínicas em modelos estatísticos ou de
          aprendizado de máquina clássico.
        \item Na base MUSIC: \cite{ramirez2017sudden, swee2026multimodal}.
      \end{itemize}
    \end{block}
    \column{0.49\textwidth}
    \begin{block}{Redes profundas sobre o sinal}
      \small
      \begin{itemize}
        \item Sucesso em diagnóstico: arritmias no Holter
          \cite{hannun2019cardiologist}, ECG de 12 derivações
          \cite{ribeiro2020automatic}.
        \item Na predição de eventos, os melhores resultados usaram redes
          \textbf{pré-treinadas} em grandes bases \cite{kim2025deep}.
      \end{itemize}
    \end{block}
  \end{columns}
\end{frame}

% 6
\begin{frame}{Desafios e lacunas}
  \begin{itemize}
    \item \textbf{Predição $\neq$ diagnóstico}: o evento ocorre meses ou anos
      depois do Holter, e as coortes têm poucas centenas de pacientes.
  \end{itemize}
  \vspace{0.2cm}
  \begin{enumerate}
    \item \textbf{Falta de comparação direta}: cada estudo avaliou uma única
      abordagem, em coortes e protocolos diferentes.
    \item \textbf{Avaliação otimista}: métricas por trecho do sinal (cada
      paciente conta centenas de vezes) e seleção de variáveis com o
      desfecho de toda a coorte \cite{kapoor2023leakage}.
    \item \textbf{Fibrilação atrial na coorte}: ritmo fácil de reconhecer e de
      pior prognóstico, que pode inflar a discriminação.
  \end{enumerate}
  \msg{Comparar as abordagens nos mesmos pacientes, com avaliação por paciente
    e em ritmo sinusal.}
\end{frame}

% 7
\begin{frame}{Objetivos}
  \begin{block}{Objetivo geral}
    \small Investigar métodos de aprendizado de máquina e de pré-processamento
    do ECG para prever a MSC em pacientes com insuficiência cardíaca crônica,
    a partir do Holter de 24 horas, em quatro anos de acompanhamento.
  \end{block}
  \begin{block}{Objetivos específicos}
    \small
    \begin{itemize}
      \item[(i)] Comparar três cenários:
        \begin{itemize}
          \item[I.] aprendizado profundo sobre o sinal bruto;
          \item[II.] pré-processamento, extração de características e
            aprendizado de máquina clássico;
          \item[III.] variáveis clínicas, isoladas e combinadas ao ECG.
        \end{itemize}
      \item[(ii)] Quantificar o ganho de combinar dados clínicos e do ECG.
      \item[(iii)] Posicionar os resultados em relação à literatura.
    \end{itemize}
  \end{block}
\end{frame}

\begin{frame}{Marcadores do ECG e clínicos na coorte MUSIC}
  \begin{block}{Ramírez et al. (2017) \cite{ramirez2017sudden}}
    \small
    \begin{itemize}
      \item 597 pacientes em ritmo sinusal, 49 mortes súbitas.
      \item Marcadores da repolarização e turbulência + clínicos, em modelo de Cox.
      \item AUC de 0,73 na FEVE reduzida, \textbf{sem validação separada}.
    \end{itemize}
  \end{block}
  \begin{block}{Swee et al. (2026) \cite{swee2026multimodal}}
    \small
    \begin{itemize}
      \item 746 pacientes, \textbf{sem restrição de ritmo}, derivação Y.
      \item Marcadores de segmentos de 30 s; sequência processada por rede
        convolucional temporal (múltiplas instâncias).
      \item AUC de 0,682 $\pm$ 0,030 apenas com o ECG (média entre partições).
    \end{itemize}
  \end{block}
\end{frame}

% 9
\begin{frame}{Aprendizado profundo no Holter e posicionamento}
  \begin{block}{Kim et al. (2025) \cite{kim2025deep}}
    \small
    \begin{itemize}
      \item 1\,108 pacientes (infarto ou insuficiência cardíaca), 70 eventos.
      \item Codificador pré-treinado e congelado; só um classificador pequeno
        é ajustado.
      \item AUC de 0,74 na coorte completa, mas \textbf{0,60} no subgrupo com
        insuficiência cardíaca.
    \end{itemize}
  \end{block}
  \begin{block}{Este trabalho}
    \small
    \begin{itemize}
      \item Três abordagens no \textbf{mesmo protocolo} e nos mesmos pacientes.
      \item Controle do otimismo: avaliação por paciente, seleção dentro da
        partição, intervalos de confiança.
      \item Restrição ao \textbf{ritmo sinusal}.
    \end{itemize}
  \end{block}
\end{frame}

\begin{frame}{Pré-processamento do ECG}
  \begin{columns}[T]
    \column{0.5\textwidth}
    \small
    \begin{itemize}
      \item \textbf{Interferências}: oscilação de linha de base ($<$ 0,5 Hz),
        ruído muscular, rede elétrica (50 Hz).
      \item \textbf{Filtro passa-faixa} Butterworth 0,5--40 Hz, aplicado nos
        dois sentidos (fase nula).
      \item \textbf{Normalização} da amplitude por derivação.
      \item \textbf{Janelas}: o registro de 24 h é dividido em trechos;
        janelas de um mesmo paciente \textbf{não são independentes}.
    \end{itemize}
    \column{0.48\textwidth}
    \centering
    \includegraphics[width=\linewidth]{figuras/ecg_tres_derivacoes.png}
    {\scriptsize\color{cinza} 10 s de Holter, derivações X, Y e Z, após o filtro.}
  \end{columns}
\end{frame}

  \begin{columns}[T]

\begin{frame}{Redes convolucionais unidimensionais}
  \small
  \begin{itemize}
    \item \textbf{Convolução 1D}: filtros que deslizam no tempo e detectam
      padrões locais (núcleo de 75 ms $\approx$ um QRS).
    \item Cada bloco: convolução $\rightarrow$ normalização em lote
      $\rightarrow$ ReLU $\rightarrow$ agrupamento máximo $\rightarrow$ descarte.
    \item \textbf{Agrupamento global} (média e máximo) e classificador.
    \item Perda: entropia cruzada ponderada, $w_{+} = N_0/N_1 \approx 8{,}4$.
  \end{itemize}
  \vspace{0.1cm}

\begin{frame}{Redes residuais}
  \begin{columns}[T]
    \column{0.5\textwidth}
    \small
    \begin{itemize}
      \item Problema: redes muito profundas ficam difíceis de treinar.
      \item \textbf{Aprendizado residual} \cite{he2016deep}: o bloco aprende
        só a diferença $\mathcal{F}(\mathbf{x})$ e soma a entrada:
        \[ \mathbf{y} = \mathrm{ReLU}\big(\mathcal{F}(\mathbf{x}) + \mathbf{x}\big) \]
      \item Atalho: identidade, ou projeção $1\times1$ quando muda a dimensão.
      \item A cada estágio: comprimento $\div 2$, canais $\times 2$.
    \end{itemize}
    \column{0.46\textwidth}
    \centering
    \begin{tikzpicture}[node distance=0.12cm,
        cv/.style={draw, rounded corners=1pt, fill=boxcolor!45, minimum width=3cm, font=\scriptsize},

\begin{frame}{Mecanismo de atenção}
  \small
  \begin{itemize}
    \item Convoluções relacionam apenas instantes \textbf{vizinhos}; ao final
      da ResNet-18, o campo receptivo é de cerca de 2,2 s.
    \item A \textbf{autoatenção} \cite{vaswani2017attention} relaciona cada
      posição com todas as outras em uma única operação:
      \[ \text{Atenção}(Q, K, V) = \mathrm{softmax}\!\left(\frac{QK^{\top}}{\sqrt{d}}\right) V \]
    \item Neste trabalho: ResNet-18 reduzida como extrator, seguida de
      \textbf{uma camada codificadora} \textit{Transformer} (4 cabeças).
    \item Custo cresce com o quadrado do comprimento da sequência.
  \end{itemize}
\end{frame}

% 15
\begin{frame}{Modelos de aprendizado de máquina clássico}
  \begin{columns}[T]
    \column{0.49\textwidth}
    \begin{block}{Regressão logística regularizada}
      \footnotesize
      \textbf{L2}: reduz os coeficientes sem zerá-los.\\
      \textbf{L1}: zera coeficientes e seleciona variáveis.
    \end{block}
    \begin{block}{Análise de componentes principais}
      \footnotesize Substitui marcadores correlacionados por poucas
      combinações não correlacionadas, sem usar o desfecho.
    \end{block}
    \column{0.49\textwidth}
    \begin{block}{Floresta aleatória \cite{breiman2001random}}
      \footnotesize Conjunto de árvores de decisão; captura relações não
      lineares e interações entre marcadores.
    \end{block}
    \begin{block}{Por que modelos simples?}
      \footnotesize 55 eventos: a regra de dez eventos por variável limita
      o modelo a poucas variáveis.
    \end{block}
  \end{columns}
\end{frame}

\begin{frame}{Avaliação de classificadores}
  \begin{columns}[T]
    \column{0.52\textwidth}
    \small
    \begin{itemize}
      \item \textbf{AUC}: probabilidade de o modelo dar risco maior a um caso
        do que a um controle. 0,5 = acaso; 1 = perfeito.
      \item \textbf{AUPRC}: área sob a curva precisão--revocação; acaso =
        prevalência (0,106) \cite{davis2006relationship}.
      \item \textbf{Agregação por paciente}: média das probabilidades das
        janelas.
      \item \textbf{Intervalo de confiança} por \textit{bootstrap} de
        pacientes.
    \end{itemize}
    \column{0.46\textwidth}
    \centering
    \begin{tikzpicture}[scale=0.85]

\begin{frame}{Pré-processamento do sinal}
  \centering

    \node[caixa, text width=2.6cm] (r) {Registro de 24 h\\X, Y, Z a 200 Hz};

    \node[caixa, text width=2.6cm, below=0.6cm of j] (f) {Butterworth\\0,5--40 Hz, fase nula};

  \begin{itemize}
    \small

\begin{frame}{Conclusões}
  \begin{itemize}
    \item \textbf{Cenário I}: redes profundas sobre o sinal bruto ficaram entre
      0,48 e 0,56, sem se distinguir do acaso; em ritmo não sinusal, a mesma
      rede aprende.
    \item \textbf{Cenário II}: os marcadores extraídos do ECG superaram as
      redes; a informação está em \textbf{poucos marcadores} ligados à
      frequência cardíaca (0,647).
    \item \textbf{Cenário III}: clínicas sozinhas, 0,661; somadas aos
      marcadores, 0,681; com os exames, até 0,738.
  \end{itemize}
  \msg{O processamento do ECG com extração de características produziu
    preditores de MSC superiores ao aprendizado profundo sobre o sinal bruto.}
\end{frame}
