\section{Introdução}

\begin{frame}{Morte súbita cardíaca: o problema}
  \begin{columns}[T]
    \begin{column}{0.56\textwidth}
      \begin{itemize}
        \item Morte de causa cardíaca, inesperada e rápida, em geral por
              arritmia ventricular
        \item Na insuficiência cardíaca, disputa espaço com a morte por
              \textbf{falência de bomba}, de mecanismo mecânico e progressivo
        \item Prevenir exige saber \textbf{quem} está em risco
              \textbf{elétrico}, e não apenas quem está mais grave
        \item O evento é raro mesmo nessa população: poucos casos para
              aprender
      \end{itemize}
    \end{column}
    \begin{column}{0.40\textwidth}
      \centering
      \begin{tikzpicture}[node distance=5mm]
        \node[caixa, minimum width=3.6cm] (ic) {Pacientes com\\insuficiência cardíaca};
        \node[caixad, below left=7mm and -14mm of ic, minimum width=1.7cm] (msc) {Morte\\súbita};
        \node[caixa, below right=7mm and -14mm of ic, minimum width=1.7cm] (fb) {Falência\\de bomba};
        \draw[seta] (ic) -- (msc);
        \draw[seta] (ic) -- (fb);
        \node[rotulo, below=1mm of msc] {elétrica};
        \node[rotulo, below=1mm of fb] {mecânica};
      \end{tikzpicture}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Por que o Holter de 24 horas?}
  \begin{columns}[c]
    \begin{column}{0.42\textwidth}
      \begin{itemize}
        \item Exame barato, não invasivo e de rotina
        \item Observa o coração ao longo do dia, e não em poucos segundos
        \item Capta ectopias, pausas e a variação da frequência entre sono e
              vigília
        \item Três derivações ortogonais (X, Y e Z), 200\,Hz: cerca de
              17 milhões de amostras por paciente
      \end{itemize}
    \end{column}
    \begin{column}{0.56\textwidth}
      \centering
      \includegraphics[width=\linewidth]{figuras/ecg_tres_derivacoes.png}\\
      {\scriptsize\color{cinza}Trecho de 10\,s de um paciente da coorte}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Objetivos e perguntas de pesquisa}
  \textbf{Objetivo geral:} investigar se o Holter de 24\,h contém informação
  para identificar, entre pacientes com insuficiência cardíaca, os que viriam
  a sofrer morte súbita.

  \vspace{0.4cm}
  \textbf{Perguntas de pesquisa}
  \begin{enumerate}
    \item Redes profundas treinadas diretamente sobre o sinal superam os
          marcadores clássicos do ECG?
    \item Aprender representações do sinal (com rótulos substitutos ou sem
          rótulos) ajuda quando há poucos eventos?
    \item O Holter acrescenta informação às variáveis clínicas e aos exames
          já medidos?
  \end{enumerate}

  \vspace{0.3cm}
  \alert{Restrição central:} 517 pacientes e apenas 55 eventos.
\end{frame}

\begin{frame}{Redes convolucionais 1D e a ResNet}
  \begin{columns}[c]
    \begin{column}{0.48\textwidth}

        \item Filtros deslizam ao longo do tempo e aprendem padrões locais

        \item Bem-sucedidas no ECG quando há \textbf{muitos} exemplos
              rotulados
      \end{itemize}

\begin{frame}{Aprendizado multitarefa e autossupervisionado}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{Multitarefa \cite{caruana1997multitask}}
        \footnotesize
        A rede aprende várias tarefas ao mesmo tempo e compartilha a
        representação entre elas.\\[3pt]
        Aqui: prever \textbf{atributos clínicos} disponíveis para quase todos
        os pacientes (QRS, FEVE, ondas Q, Pro-BNP\ldots) como
        \textbf{rótulos substitutos}.
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Autossupervisionado \cite{chen2020simple}}
        \footnotesize
        A rede aprende sem rótulos, a partir do próprio sinal.\\[3pt]
        \textbf{Contrastivo}: janelas do mesmo paciente devem ter
        representações próximas; de pacientes diferentes, distantes.
      \end{block}
    \end{column}
  \end{columns}

  \vspace{0.5cm}
  \centering
  \begin{tikzpicture}[node distance=6mm]
    \node[caixa] (s) {Sinal};
    \node[caixa, right=of s] (r) {Representação\\aprendida};
    \node[caixad, right=of r] (m) {Classificador simples\\(55 eventos)};
    \draw[seta] (s) -- node[above, rotulo]{rede} (r);
    \draw[seta] (r) -- (m);
  \end{tikzpicture}

  {\footnotesize Ideia comum: a rede aprende com muitos rótulos ou sem
  rótulos; o desfecho raro entra só no fim.}
\end{frame}

\begin{frame}{Abordagem 1: biomarcadores e regressão logística}
  \centering
  \begin{tikzpicture}[node distance=5mm]
    \node[caixa] (h) {Holter\\24\,h};
    \node[caixa, right=of h] (s) {Segmentos de 30\,s\\a cada 2 min};
    \node[caixa, right=of s] (m) {Marcadores\\(NeuroKit2)};
    \node[caixa, right=of m] (p) {Resumo por\\paciente};
    \node[caixad, right=of p] (l) {Regressão\\logística};
    \draw[seta] (h) -- (s);
    \draw[seta] (s) -- (m);
    \draw[seta] (m) -- (p);
    \draw[seta] (p) -- (l);
  \end{tikzpicture}

  \vspace{0.5cm}
  \raggedright
  \begin{itemize}
    \item Segmentos com menos de 80\,\% de batimentos normais são descartados
    \item Por segmento, usa-se a derivação mais limpa
    \item \textbf{Modelo pré-especificado}: dois marcadores escolhidos pela
          literatura, e não pelos dados
          \begin{itemize}
            \item variação da frequência cardíaca ao longo do dia
            \item carga de ectopia
          \end{itemize}
    \item Dois protocolos de extração: \textbf{A}, segmentos de 5 min com
          44 marcadores; \textbf{B}, segmentos de 30\,s
    \item Escolher os marcadores olhando o desfecho de toda a coorte
          superestima o desempenho \cite{kapoor2023leakage}
  \end{itemize}
\end{frame}
