\begin{frame}{Morte súbita cardíaca: quando o coração para sem aviso}
  Na insuficiência cardíaca, a morte chega por duas vias: a
  \textbf{falência de bomba}, lenta e progressiva, e a \textbf{morte súbita
  cardíaca (MSC)}, abrupta.
  \vspace{0.4em}
  \begin{columns}[T]
    \column{0.32\textwidth}
    \centering
    \textbf{O que é}\\[2pt]
    \small evento abrupto, em geral por arritmia ventricular, em paciente
    muitas vezes estável
    \column{0.32\textwidth}
    \centering
    \textbf{Quanto pesa}\\[2pt]
    \small cerca de \textbf{metade} das mortes na insuficiência cardíaca
    \cite{tomaselli2004sudden}
    \column{0.32\textwidth}
    \centering
    \textbf{Como se previne}\\[2pt]
    \small cardiodesfibrilador implantável, indicado pela fração de ejeção
    (FEVE) \cite{bardy2005amiodarone}
  \end{columns}
  \vspace{0.8em}
  \begin{block}{O que esses fatos têm em comum?}
    A MSC nasce de uma \textbf{falha elétrica} do coração --- e é justamente
    a atividade elétrica que o ECG registra. O Holter de 24 horas, que a
    acompanha por um dia inteiro, já guardaria sinais desse risco?
  \end{block}
\end{frame}

% 4
\begin{frame}{O que queremos prever?}
  \begin{columns}[c]
    \column{0.58\textwidth}
    \small
    \begin{block}{Classificar quem terá MSC em 4 anos (sim ou não)}
      \begin{itemize}
        \item O modelo estima uma \textbf{probabilidade} de MSC para cada
          paciente.
        \item A \textbf{AUC} avalia essa probabilidade: os casos recebem
          risco maior que os controles?
        \item Um \textbf{limiar} transforma a probabilidade em sim ou não,
          para calcular sensibilidade e especificidade.
      \end{itemize}
    \end{block}
    \begin{block}{A partir do Holter de 24 horas}
      \begin{itemize}
        \item Exame não invasivo, barato e amplamente disponível.
        \item Registra ectopias, pausas e a variação da frequência entre sono
          e vigília.
      \end{itemize}
    \end{block}
    \column{0.39\textwidth}
    \centering
    \begin{tikzpicture}[node distance=0.35cm]
      \node[caixa, text width=3.6cm] (h) {Holter de 24 h\\3 derivações, 200 Hz};
      \node[caixad, text width=3.6cm, below=of h] (m) {Modelo de\\aprendizado de máquina};
      \node[caixav, text width=3.6cm, below=of m] (p) {Probabilidade de MSC\\em 4 anos};
      \node[caixa, text width=3.6cm, below=of p] (c) {Limiar $\rightarrow$\\\textbf{sim} ou \textbf{não}};
      \draw[seta] (h) -- (m);
      \draw[seta] (m) -- (p);
      \draw[seta] (p) -- (c);
    \end{tikzpicture}
  \end{columns}
\end{frame}

% 5
\begin{frame}{Por que a fração de ejeção não basta}
  Hoje, a decisão de implantar o cardiodesfibrilador depende basicamente de
  \textbf{um único número}: a FEVE abaixo de um limiar
  \cite{zeppenfeld2022esc}.
  \vspace{0.4em}
  \begin{columns}[T]
    \column{0.48\textwidth}
    \begin{block}{Quem recebe o dispositivo}
      Muitos pacientes nunca chegam a utilizá-lo: o critério indica
      pessoas que não teriam a arritmia.
    \end{block}
    \column{0.48\textwidth}
    \begin{block}{Quem fica de fora}
      Parte das mortes súbitas ocorre com FEVE \textbf{acima} do limiar: o
      critério não alcança esses pacientes.
    \end{block}
  \end{columns}
  \vspace{0.6em}
  \begin{block}{Por que isso importa}
    Há espaço para ferramentas que \textbf{estratifiquem melhor} o risco de
    MSC --- e o ECG, que mede diretamente a atividade elétrica, é um
    candidato natural.
  \end{block}
\end{frame}

% 6
\begin{frame}{Como a literatura tem abordado o problema}
  Os trabalhos se organizam em duas frentes, raramente comparadas entre si:
  \vspace{0.3em}
  \begin{columns}[T]
    \column{0.49\textwidth}
    \begin{block}{Marcadores definidos por especialistas}
      \begin{itemize}
        \item VFC, ectopia, turbulência, repolarização.
        \item Combinados com variáveis clínicas em modelos estatísticos.
        \item Na base MUSIC: \cite{ramirez2017sudden, swee2026multimodal}.
      \end{itemize}
    \end{block}
    \column{0.49\textwidth}
    \begin{block}{Redes profundas sobre o sinal}
      \begin{itemize}
        \item Sucesso no \textbf{diagnóstico} de arritmias
          \cite{hannun2019cardiologist, ribeiro2020automatic}.
        \item Na \textbf{predição} de eventos, os melhores resultados usaram
          redes pré-treinadas em grandes bases \cite{kim2025deep}.
      \end{itemize}
    \end{block}
  \end{columns}
  \vspace{0.5em}
  \begin{block}{Prever não é diagnosticar}
    O evento ocorre meses ou anos depois do exame, e as coortes têm poucas
    centenas de pacientes.
  \end{block}
\end{frame}

% 7
\begin{frame}{Objetivo e contribuição deste trabalho}
  \small
  \begin{block}{Objetivo geral}
    Investigar métodos de aprendizado de máquina e de pré-processamento do
    ECG para prever a MSC em pacientes com insuficiência cardíaca crônica, a
    partir do Holter de 24 horas, em quatro anos de acompanhamento.
  \end{block}
  Três lacunas motivam o trabalho:
  \begin{enumerate}\footnotesize
    \item \textbf{falta de comparação direta}: cada estudo avaliou uma única
      abordagem, em coortes e protocolos diferentes;
    \item \textbf{avaliação otimista}: métricas por trecho do sinal e seleção
      de variáveis com o desfecho de toda a coorte \cite{kapoor2023leakage};
    \item \textbf{fibrilação atrial na coorte}, que pode inflar a
      discriminação.
  \end{enumerate}
  \vspace{0.2em}
  \begin{block}{Contribuição: três cenários nos mesmos 517 pacientes}
    \textbf{I.} redes profundas sobre o sinal bruto; \textbf{II.} marcadores
    extraídos do ECG com aprendizado de máquina clássico; \textbf{III.}
    variáveis clínicas, isoladas e combinadas ao ECG --- com avaliação por
    paciente e em ritmo sinusal.
  \end{block}
\end{frame}

\begin{frame}{Na coorte MUSIC: marcadores do ECG e variáveis clínicas}
  \begin{block}{\citeonline{ramirez2017sudden}}
    \small
    \begin{itemize}
      \item 597 pacientes em ritmo sinusal, 49 mortes súbitas.
      \item Marcadores da repolarização e turbulência + clínicos, em modelo de Cox.
      \item AUC de 0,73 na FEVE reduzida, \textbf{sem validação separada}.
    \end{itemize}
  \end{block}
  \begin{block}{\citeonline{swee2026multimodal}}
    \small
    \begin{itemize}
      \item 746 pacientes, \textbf{sem restrição de ritmo}, derivação Y.
      \item Marcadores de segmentos de 30 s; sequência processada por rede
        convolucional temporal (múltiplas instâncias).
      \item AUC de 0,682 $\pm$ 0,030 apenas com o ECG (média entre partições).
    \end{itemize}
  \end{block}
\end{frame}

% 9
\begin{frame}{Redes profundas no Holter, e onde este trabalho se encaixa}
  \begin{block}{\citeonline{kim2025deep}}
    \small
    \begin{itemize}
      \item 1\,108 pacientes (infarto ou insuficiência cardíaca), 70 eventos.
      \item Codificador pré-treinado e congelado; só um classificador pequeno
        é ajustado.
      \item AUC de 0,74 na coorte completa, mas \textbf{0,60} no subgrupo com
        insuficiência cardíaca.
    \end{itemize}
  \end{block}
  \begin{block}{Este trabalho}
    \small
    \begin{itemize}
      \item Três abordagens no \textbf{mesmo protocolo} e nos mesmos pacientes.
      \item Controle do otimismo: avaliação por paciente, seleção dentro da
        partição, intervalos de confiança.
      \item Restrição ao \textbf{ritmo sinusal}.
    \end{itemize}
  \end{block}
\end{frame}

\begin{frame}{Antes de analisar: limpar o sinal do ECG}
  \begin{columns}[T]
    \column{0.5\textwidth}
    \small
    \begin{itemize}
      \item \textbf{Interferências}: oscilação de linha de base ($<$ 0,5 Hz),
        ruído muscular, rede elétrica (50 Hz).
      \item \textbf{Filtro passa-faixa} Butterworth 0,5--40 Hz, aplicado nos
        dois sentidos (fase nula).
      \item \textbf{Normalização} da amplitude por derivação.
      \item \textbf{Janelas}: o registro de 24 h é dividido em trechos;
        janelas de um mesmo paciente \textbf{não são independentes}.
    \end{itemize}
    \column{0.48\textwidth}
    \centering
    \includegraphics[width=\linewidth]{figuras/ecg_tres_derivacoes.png}
    {\scriptsize\color{cinza} 10 s de Holter, derivações X, Y e Z, após o filtro.}
  \end{columns}
\end{frame}

\begin{frame}{Redes convolucionais: aprender padrões direto do sinal}
  \small
  \begin{itemize}
    \item \textbf{Convolução 1D}: filtros que deslizam no tempo e detectam
      padrões locais (núcleo de 75 ms $\approx$ um QRS).
    \item Cada bloco: convolução $\rightarrow$ normalização em lote
      $\rightarrow$ ReLU $\rightarrow$ agrupamento máximo $\rightarrow$ descarte.
    \item \textbf{Agrupamento global} (média e máximo) e classificador.
    \item Perda: entropia cruzada ponderada, $w_{+} = N_0/N_1 \approx 8{,}4$.
  \end{itemize}
  \fonte{A arquitetura completa, com as dimensões de cada bloco, está na Metodologia (Cenário I).}
\end{frame}

% 13
\begin{frame}{Redes residuais: como treinar redes mais profundas}
  \small
  \begin{itemize}
    \item Problema: redes muito profundas ficam difíceis de treinar.
    \item \textbf{Aprendizado residual} \cite{he2016deep}: o bloco aprende
      só a diferença $\mathcal{F}(\mathbf{x})$ e soma a entrada:
      \equacao{\mathbf{y} = \mathrm{ReLU}\big(\mathcal{F}(\mathbf{x}) + \mathbf{x}\big)}
    \item Atalho: identidade, ou projeção $1\times1$ quando muda a dimensão.
    \item Dois tipos de bloco neste trabalho: \textbf{básico} (duas convoluções)
      e de \textbf{gargalo} (três convoluções, mais barato em redes profundas).
    \item A cada estágio da rede: comprimento $\div 2$ e canais $\times 2$.
  \end{itemize}
\end{frame}

\begin{frame}{Atenção: relacionar instantes distantes do registro}
  \small
  \begin{itemize}
    \item Convoluções relacionam apenas instantes \textbf{vizinhos}; ao final
      da ResNet-18, o campo receptivo é de cerca de 2,2 s.
    \item A \textbf{autoatenção} \cite{vaswani2017attention} relaciona cada
      posição com todas as outras em uma única operação:
      \equacao{\text{Atenção}(Q, K, V) = \mathrm{softmax}\!\left(\frac{QK^{\top}}{\sqrt{d}}\right) V}
    \item Neste trabalho: ResNet-18 reduzida como extrator, seguida de
      \textbf{uma camada codificadora} \textit{Transformer} (4 cabeças).
    \item Custo cresce com o quadrado do comprimento da sequência.
  \end{itemize}
\end{frame}

\begin{frame}{Modelos clássicos: poucas variáveis, poucos eventos}
  \begin{columns}[T]
    \column{0.49\textwidth}
    \begin{block}{Regressão logística regularizada}
      \footnotesize
      \textbf{L2}: reduz os coeficientes sem zerá-los.\\
      \textbf{L1}: zera coeficientes e seleciona variáveis.
    \end{block}
    \begin{block}{Análise de componentes principais}
      \footnotesize Substitui marcadores correlacionados por poucas
      combinações não correlacionadas, sem usar o desfecho.
    \end{block}
    \column{0.49\textwidth}
    \begin{block}{Floresta aleatória \cite{breiman2001random}}
      \footnotesize Conjunto de árvores de decisão; captura relações não
      lineares e interações entre marcadores.
    \end{block}
    \begin{block}{Por que modelos simples?}
      \footnotesize 55 eventos: a regra de dez eventos por variável limita
      o modelo a poucas variáveis.
    \end{block}
  \end{columns}
\end{frame}

\begin{frame}{Como medir se o modelo separa casos de controles?}
  \begin{columns}[T]
    \column{0.52\textwidth}
    \small
    \begin{itemize}
      \item \textbf{AUC}: probabilidade de o modelo dar risco maior a um caso
        do que a um controle. 0,5 = acaso; 1 = perfeito.
      \item \textbf{AUPRC}: área sob a curva precisão--revocação; acaso =
        prevalência (0,106) \cite{davis2006relationship}.
      \item \textbf{Agregação por paciente}: média das probabilidades das
        janelas.
      \item \textbf{Intervalo de confiança} por \textit{bootstrap} de
        pacientes.
    \end{itemize}
    \column{0.46\textwidth}
    \centering
    \begin{tikzpicture}[scale=0.85]

\end{frame}

\begin{frame}{A base MUSIC: 992 pacientes acompanhados por quatro anos}
  \begin{itemize}
    \item Pública no PhysioNet \cite{martinyebra2025music, martinyebra2024music}.
    \item \textbf{992 pacientes} com insuficiência cardíaca crônica (NYHA II e
      III), recrutados em 2003--2004 em oito hospitais da Espanha.

\section{Conclusões}

\begin{frame}{O que aprendemos}
  \begin{itemize}
    \item \textbf{Cenário I}: redes profundas sobre o sinal bruto ficaram entre
      0,48 e 0,56, sem se distinguir do acaso; em ritmo não sinusal, a mesma
      rede aprende.
    \item \textbf{Cenário II}: os marcadores extraídos do ECG superaram as
      redes; a informação está em \textbf{poucos marcadores} ligados à
      frequência cardíaca (0,647).
    \item \textbf{Cenário III}: clínicas sozinhas, 0,661; somadas aos
      marcadores, 0,681; com os exames, até 0,738.
  \end{itemize}
  \msg{O processamento do ECG com extração de características produziu
    preditores de MSC superiores ao aprendizado profundo sobre o sinal bruto.}
\end{frame}
